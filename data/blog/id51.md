Examples:

```c
//printing in for loop

#include <stdio.h>

int main()
{
     int i;
     for (i = 0; i < 10; i++)
           printf("i = %d\n", i);

     getchar();
     return 0;
}
```

```c
//reverse for loop

#include <stdio.h>

int main()
{
     int i;
     for (i = 10; i > 0; i--)
           printf("i = %d\n", i);

     getchar();
     return 0;
}
```

```c
//infinite for loop

#include <stdio.h>

int main()
{
     for (;;)
           printf("this is an infinite loop\n");
           //once you enter in this for loop you will never return

     return 0;
}
```

```c
//for Loop Variations

#include<stdio.h>

int main()
{
     int i, j;

     for (i = 0, j = 10; i<10;   i++, j--)
           printf("%d + %d = %d\n", i, j, i + j);

     getchar();
     return 0;
}
```

```c
//for Loops with No Bodies

#include <stdio.h>

int main(void)
{
     int n = 10;
     int k = 1;
     int i;

     for (i = 1, k; i <= n; k *=   i, i++);

     printf("%d", k);

     getchar();
     return 0;
}
```

```c
//Declaring Variables within a for Loop

#include <stdio.h>

int main()
{
     int i;

     for (i = 1; i < 10; i++)
     {
           int sq_i = i*i;
           int qb_i = i*i*i;
           printf("square of %d = %d\n", i, sq_i);
           printf("qube of %d = %d\n", i, qb_i);
     }

     getchar();
     return 0;
}
```
