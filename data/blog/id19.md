```c
#include <stdio.h>
#include <conio.h>

int main()
{
     char password[20], ch = 0;
     int i = 0, j;
     printf("Enter the password <limit 20 characters>: ");
     while (i < 20)
     {
           ch = _getch();
           if (ch == `\r`)
                break;
           password[i] = ch;
           i++;
           ch = `*`;
           printf("%c", ch);
     }
     password[i] = `\0`;

     printf("\nThe password is :");
     for (j = 0; j < i; j++)
     {
           printf("%c", password[j]);
     }
     return 0;
}
```
